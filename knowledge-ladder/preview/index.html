暴力枚举
数据范围小或答案空间有限时，直接枚举所有情况并验证。
复杂度：O(n²)（本例）
int maxSubarray(const vector<int>& a) {
  int n = (int)a.size(), ans = INT_MIN;
  for (int l = 0; l < n; l++) {
    int s = 0;
    for (int r = l; r < n; r++) {
      s += a[r];
      ans = max(ans, s);
    }
  }
  return ans;
}
